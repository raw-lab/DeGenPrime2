@echo off
REM DeGenPrime installer for Windows.
REM Rename this file to install.bat first: it ships as .txt so that e-mail
REM filters let the zip through. Needs Rust from https://rustup.rs
setlocal
cd /d "%~dp0"
REM rustup installs cargo here; a window opened before Rust was installed
REM does not have it on its PATH yet.
if exist "%USERPROFILE%\.cargo\bin\cargo.exe" set "PATH=%USERPROFILE%\.cargo\bin;%PATH%"
where cargo >nul 2>nul
if errorlevel 1 (
  echo Rust is not installed. Install it from https://rustup.rs and run this file again.
  pause
  exit /b 1
)
echo Building DeGenPrime...
cargo build --release --locked
if errorlevel 1 (
  echo Build failed. If the message mentions link.exe, install the Visual Studio C++ Build Tools
  echo ^(the Rust installer offers them^) and run this file again.
  pause
  exit /b 1
)
set "DEST=%LOCALAPPDATA%\DeGenPrime"
if not exist "%DEST%" mkdir "%DEST%"
copy /y target\release\degenprime.exe "%DEST%\degenprime.exe" >nul
if errorlevel 1 (
  echo Could not replace degenprime.exe. Close DeGenPrime if it is running and run this file again.
  pause
  exit /b 1
)
copy /y assets\icon.ico "%DEST%\degenprime.ico" >nul
REM Start Menu and desktop shortcuts that open the desktop app. The paths come
REM from the environment, so a user name with quotes or spaces cannot break them.
powershell -NoProfile -ExecutionPolicy Bypass -Command "$s = New-Object -ComObject WScript.Shell; foreach ($d in @([Environment]::GetFolderPath('Programs'), [Environment]::GetFolderPath('Desktop'))) { $l = $s.CreateShortcut((Join-Path $d 'DeGenPrime.lnk')); $l.TargetPath = (Join-Path $env:DEST 'degenprime.exe'); $l.Arguments = 'gui'; $l.IconLocation = (Join-Path $env:DEST 'degenprime.ico'); $l.WorkingDirectory = $env:DEST; $l.Description = 'Degenerate PCR primer design'; $l.Save() }"
REM Put degenprime on the user PATH if it is not there yet.
powershell -NoProfile -ExecutionPolicy Bypass -Command "$p = [Environment]::GetEnvironmentVariable('Path', 'User'); if ($null -eq $p) { $p = '' }; if (-not (($p -split ';') -contains $env:DEST)) { $n = (@($p.TrimEnd(';'), $env:DEST) | Where-Object { $_ }) -join ';'; [Environment]::SetEnvironmentVariable('Path', $n, 'User') }"
"%DEST%\degenprime.exe" doctor
echo.
echo Done. Open DeGenPrime from the Start menu or the desktop, or run: degenprime gui
echo Open a new terminal window first, so that it sees the updated PATH.
pause
